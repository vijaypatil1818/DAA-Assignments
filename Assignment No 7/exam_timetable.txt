
import java.util.*;

public class GraphColoring {

    static String[] colorNames = {
        "Red", "Green", "Yellow", "Blue",
        "Magenta", "Cyan", "Orange", "Purple"
    };

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        System.out.print("Enter number of courses: ");
        int n = sc.nextInt();

        System.out.print("Enter number of students: ");
        int m = sc.nextInt();

        if (n <= 0 || m < 0) {
            System.out.println("Invalid number of courses or students!");
            sc.close();
            return;
        }

        boolean[][] conflict = new boolean[n][n];

        System.out.println("\nEnter courses for each student.");
        System.out.println(
            "Enter the number of courses followed by course IDs (1 to "
            + n + ")."
        );

        // Input courses and build the conflict graph
        for (int i = 0; i < m; i++) {
            System.out.print(
                "\nStudent " + (i + 1) + " - number of courses: "
            );
            int k = sc.nextInt();

            if (k < 0 || k > n) {
                System.out.println("Invalid number of courses!");
                sc.close();
                return;
            }

            System.out.print("Enter course IDs: ");

            boolean[] enrolled = new boolean[n];

            for (int j = 0; j < k; j++) {
                int course = sc.nextInt();

                if (course < 1 || course > n) {
                    System.out.println("Invalid course ID!");
                    sc.close();
                    return;
                }

                enrolled[course - 1] = true;
            }

            // Courses taken by the same student conflict
            for (int u = 0; u < n; u++) {
                for (int v = u + 1; v < n; v++) {
                    if (enrolled[u] && enrolled[v]) {
                        conflict[u][v] = true;
                        conflict[v][u] = true;
                    }
                }
            }
        }

        // Calculate the degree of each course
        List<int[]> courses = new ArrayList<>();

        for (int i = 0; i < n; i++) {
            int degree = 0;

            for (int j = 0; j < n; j++) {
                if (conflict[i][j]) {
                    degree++;
                }
            }

            // Store {degree, course index}
            courses.add(new int[]{degree, i});
        }

        // Welsh-Powell: sort by decreasing degree.
        // Break ties by course index for predictable output.
        courses.sort((a, b) -> {
            if (a[0] != b[0]) {
                return Integer.compare(b[0], a[0]);
            }
            return Integer.compare(a[1], b[1]);
        });

        // Greedy graph coloring
        int[] color = new int[n];
        Arrays.fill(color, -1);

        int totalSlots = 0;

        for (int[] course : courses) {
            int u = course[1];
            boolean[] used = new boolean[n];

            // Find colors already used by neighboring courses
            for (int v = 0; v < n; v++) {
                if (conflict[u][v] && color[v] != -1) {
                    used[color[v]] = true;
                }
            }

            // Assign the smallest available color
            int c = 0;
            while (c < n && used[c]) {
                c++;
            }

            color[u] = c;
            totalSlots = Math.max(totalSlots, c + 1);
        }

        // Display examination timetable
        System.out.println("\n--- Examination Timetable ---");
        System.out.printf("%-10s %-12s %-10s%n",
                          "Course", "Time Slot", "Color");

        for (int i = 0; i < n; i++) {
            System.out.printf("%-10s %-12s %-10s%n",
                "C" + (i + 1),
                "Slot " + (color[i] + 1),
                colorNames[color[i] % colorNames.length]
            );
        }

        System.out.println(
            "\nTotal time slots required: " + totalSlots
        );

        // Display graph coloring
        System.out.println("\n--- Graph Coloring ---");

        for (int i = 0; i < n; i++) {
            System.out.println(
                "[ C" + (i + 1) + " ] --> Slot "
                + (color[i] + 1)
                + " | Color: "
                + colorNames[color[i] % colorNames.length]
            );
        }

        // Display conflict graph edges
        System.out.println("\n--- Conflict Graph Edges ---");

        boolean hasEdges = false;

        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (conflict[i][j]) {
                    System.out.println(
                        "C" + (i + 1) + " -- C" + (j + 1)
                    );
                    hasEdges = true;
                }
            }
        }

        if (!hasEdges) {
            System.out.println("No conflicts between courses.");
        }

        sc.close();
    }
}
